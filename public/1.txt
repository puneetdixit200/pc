/*
Program 1 - Sequential and Parallel Merge Sort (OpenMP)
Compile: gcc -x c -fopenmp 1.txt -o p1
Run:     ./p1
Input:   100000 (type after starting the program)
*/

#include <stdio.h>
#include <stdlib.h>
#include <omp.h>

void merge(int a[], int l, int m, int r) {
    int n = r - l + 1;
    int *temp = malloc(n * sizeof(int));
    int i = l, j = m + 1, k = 0;

    while (i <= m && j <= r)
        temp[k++] = a[i] <= a[j] ? a[i++] : a[j++];

    while (i <= m) temp[k++] = a[i++];
    while (j <= r) temp[k++] = a[j++];

    for (i = 0; i < n; i++)
        a[l + i] = temp[i];

    free(temp);
}

void seq(int a[], int l, int r) {
    if (l >= r) return;

    int m = (l + r) / 2;
    seq(a, l, m);
    seq(a, m + 1, r);
    merge(a, l, m, r);
}

void par(int a[], int l, int r, int depth) {
    if (l >= r) return;

    int m = (l + r) / 2;

    if (depth == 0) {
        seq(a, l, m);
        seq(a, m + 1, r);
    } else {
        #pragma omp parallel sections
        {
             #pragma omp section
             par(a, l, m, depth - 1);

            #pragma omp section
            par(a, m + 1, r, depth - 1);
        }
    }

    merge(a, l, m, r);
}

int main() {
    int n;
    printf("Enter array size: ");
      if (scanf("%d", &n) != 1 || n <= 0) return 1;

      int *a = malloc(n * sizeof(int));
      int *b = malloc(n * sizeof(int));
      if (!a || !b) return 1;

      for (int i = 0; i < n; i++)
          a[i] = b[i] = rand() % 100000;

      double t = omp_get_wtime();
      seq(a, 0, n - 1);
      printf("Sequential Merge Sort Time: %f seconds\n", omp_get_wtime() - t);

      omp_set_max_active_levels(4);
      t = omp_get_wtime();
      par(b, 0, n - 1, 3);
      printf("Parallel Merge Sort Time: %f seconds\n", omp_get_wtime() - t);

      free(a);
      free(b);
      return 0;
 }
