/*
Program 8 - MPI Scatter and Gather
Compile: mpicc -x c 8.txt -o p8
Run:     mpirun -np 10 ./p8
If insufficient CPU slots: mpirun --oversubscribe -np 10 ./p8
*/

#include <stdio.h>
#include <mpi.h>

int main(int argc, char **argv) {
    int rank, size, send_data[100], gathered_data[100];

    MPI_Init(&argc, &argv);
    MPI_Comm_rank(MPI_COMM_WORLD, &rank);
    MPI_Comm_size(MPI_COMM_WORLD, &size);

    if (100 % size != 0) {
        if (rank == 0)
            printf("Array size (100) must be divisible by number of processes.\n");
        MPI_Finalize();
        return 0;
    }

    int chunk = 100 / size;
    int recv_data[chunk];

    if (rank == 0) {
        for (int i = 0; i < 100; i++)
            send_data[i] = i + 1;
        printf("Root process initialized array to scatter.\n");
    }

    MPI_Scatter(send_data, chunk, MPI_INT, recv_data, chunk, MPI_INT, 0, MPI_COMM_WORLD);

    printf("Process %d received data: ", rank);
    for (int i = 0; i < chunk; i++) {
        printf("%d ", recv_data[i]);
        recv_data[i] *= 2;
    }
    printf("\n");

    MPI_Gather(recv_data, chunk, MPI_INT, gathered_data, chunk, MPI_INT, 0, MPI_COMM_WORLD);

    if (rank == 0) {
        printf("Root process gathered modified data: ");
        for (int i = 0; i < 100; i++)
            printf("%d ", gathered_data[i]);
        printf("\n");
    }

    MPI_Finalize();
    return 0;
}
