/*
Program 2 - OpenMP Static Scheduling, Chunk Size 2
Compile: gcc -x c -fopenmp 2.txt -o p2
Run:     OMP_NUM_THREADS=2 ./p2 4
*/

#include <stdio.h>
#include <stdlib.h>
#include <omp.h>

int main(int argc, char *argv[]) {
    if (argc != 2) {
        printf("Usage: %s <iterations>\n", argv[0]);
        return 1;
    }

    int n = atoi(argv[1]);
    int *owner = malloc(n * sizeof(int));
    if (!owner) return 1;

    #pragma omp parallel for schedule(static, 2)
    for (int i = 0; i < n; i++)
        owner[i] = omp_get_thread_num();

    for (int i = 0; i < n; i += 2)
        printf("Thread %d: Iterations %d -- %d\n",
               owner[i], i, i + 1 < n ? i + 1 : i);

    free(owner);
    return 0;
}
