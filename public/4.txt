/*
Program 4 - Prime Numbers using OpenMP Parallel For
Compile: gcc -x c -fopenmp 4.txt -o p4
Run:     ./p4 100000
*/

#include <stdio.h>
#include <stdlib.h>
#include <omp.h>

int is_prime(int n) {
    if (n < 2) return 0;
    if (n == 2) return 1;
    if (n % 2 == 0) return 0;

    for (int i = 3; i <= n / i; i += 2)
        if (n % i == 0) return 0;

    return 1;
}

int main(int argc, char *argv[]) {
    if (argc != 2) {
        printf("Usage: %s <n>\n", argv[0]);
        return 1;
    }

    int n = atoi(argv[1]);
    int first[10], count1 = 0, count2 = 0;

    double t = omp_get_wtime();
    for (int i = 1; i <= n; i++) {
        if (is_prime(i)) {
            if (count1 < 10) first[count1] = i;
            count1++;
        }
    }
    double serial = omp_get_wtime() - t;

    t = omp_get_wtime();
    #pragma omp parallel for schedule(static)
    for (int i = 1; i <= n; i++) {
        if (is_prime(i)) {
            #pragma omp critical
            count2++;
        }
    }
    double parallel = omp_get_wtime() - t;

    printf("Serial Time : %f seconds\n", serial);
    printf("Parallel Time : %f seconds\n", parallel);
    printf("Number of primes found: %d\n", count1);
    printf("First 10 primes (serial): ");
    for (int i = 0; i < count1 && i < 10; i++)
        printf("%d ", first[i]);
    printf("\n");

    return count1 != count2;
}
